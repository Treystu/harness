   268:     def order(self, models: Iterable[str], *, prefer_free: bool = True) -> List[str]:
   269:         """Servable models in the caller's order, free routes first (#540).
   270: 
   271:         Excluded pairs are dropped rather than merely demoted: dispatching to

   273:         When every model is excluded the caller's order is returned unchanged
   274:         -- a ladder must never become empty because of bookkeeping.

   666: def heuristic_decompose_goal(goal: str, candidate_files: Optional[Sequence[str]] = None,
   667:                              *, root_dir: Optional[str] = None) -> TaskDAG:
   668:     # Deterministic heuristic decomposition into a TaskDAG.
   669:     # #456 greenfield creation: candidate files that do not exist under
   670:     # root_dir are marked create_files, so the executor creates (not edits)
   671:     # them. Without root_dir every candidate is assumed to exist.
   672:     if not goal or not goal.strip():
   673:         raise HarnessError("goal cannot be empty")
   674: 
   675:     files = [f.strip() for f in (candidate_files or []) if str(f).strip()]
   676:     nodes: Dict[str, DAGNode] = {}
   677: 
   678:     def _creates(targets):
   679:         if root_dir is None:
   680:             return ()
   681:         missing = []
   682:         for rel in targets:
   683:             full = rel if os.path.isabs(rel) else os.path.join(root_dir, rel)
   684:             if not os.path.exists(full):
   685:                 missing.append(rel)
   686:         return tuple(missing)
   687: 
   688:     # Check for numbered or bulleted steps in goal
   689:     steps = [s.strip() for s in re.split(r"(?:^|\n)\s*(?:\d+[\.\)]|[-*])\s+", goal.strip()) if s.strip()]
   690: 
   691:     if len(steps) > 1:
   692:         # Multi-step goal
   693:         prev_id = None
   694:         for i, step in enumerate(steps, 1):
   695:             node_id = f"task_{i}"
   696:             deps = (prev_id,) if prev_id else ()
   697:             target = (files[i - 1],) if i - 1 < len(files) else ()
   698:             nodes[node_id] = DAGNode(
   699:                 node_id=node_id,
   700:                 instruction=step,
   701:                 target_files=target,
   702:                 create_files=_creates(target),
   703:                 dependencies=deps,
   704:                 complexity_tier=1,
   705:             )
   706:             prev_id = node_id
   707:     elif len(files) > 1:
   708:         # Multi-file goal: one subtask per file
   709:         test_files = [f for f in files if "test" in f.lower()]
   710:         impl_files = [f for f in files if f not in test_files]
   711: 
   712:         impl_ids = []
   713:         for i, f in enumerate(impl_files, 1):
   714:             node_id = f"task_{i}"
   715:             impl_ids.append(node_id)
   716:             nodes[node_id] = DAGNode(
   717:                 node_id=node_id,
   718:                 instruction=f"{goal.strip()} for {f}",
   719:                 target_files=(f,),
   720:                 create_files=_creates((f,)),
   721:                 dependencies=(),
   722:                 complexity_tier=1,
   723:             )
   724: 
   725:         start_test_idx = len(impl_files) + 1
   726:         for j, f in enumerate(test_files, start_test_idx):
   727:             node_id = f"task_{j}"
   728:             nodes[node_id] = DAGNode(
   729:                 node_id=node_id,
   730:                 instruction=f"Update tests in {f} for {goal.strip()}",
   731:                 target_files=(f,),
   732:                 create_files=_creates((f,)),
   733:                 dependencies=tuple(impl_ids),
   734:                 complexity_tier=1,
   735:             )
   736:     else:
   737:         # Single node goal
   738:         target = tuple(files)
   739:         if not target:
   740:             # If no files were provided, check if the goal mentions a specific file
   741:             found = re.findall(r"\b[\w-]+\.(?:py|rs|go|ts|js|md|json|toml|yaml|yml|c|cpp|h)\b", goal)
   742:             if found:
   743:                 target = (found[0],)
   744:             else:
   745:                 # Default deliverable target for analytical / open-ended tasks
   746:                 target = ("docs/reports/analysis.md",)
   747:         nodes["task_1"] = DAGNode(
   748:             node_id="task_1",
   749:             instruction=goal.strip(),
   750:             target_files=target,
   751:             create_files=_creates(target),
   752:             dependencies=(),
   753:             complexity_tier=1,
   754:         )
   755: 
   756:     return TaskDAG(nodes=nodes)

   792: def _call_worst_case(governor, model, max_tokens) -> float:
   793:     """Worst-case dollar cost of ONE bounded chat call on ``model``.
   794: 
   795:     Completion-side only (prompt tokens are typically small relative to the
   796:     pinned output budget for plan-lane calls). Free / unpriceable models
   797:     contribute $0.0 -- the node ceilings still bind the run.
   798:     """
   799:     if governor is None or not model:
   800:         return 0.0
   801:     try:
   802:         pricing = governor.fetch_pricing([model])
   803:         _pp, cp = pricing[model]
   804:     except Exception:
   805:         return 0.0
   806:     try:
   807:         return max(0.0, float(max_tokens) * float(cp))
   808:     except (TypeError, ValueError):
   809:         return 0.0

   812: def composed_worst_case(

   825: 
   826:     ONE owner of the pre-execute worst-case budget. Called before execute
   827:     spend when a governor is present; if the composed total exceeds
   828:     ``governor.remaining()`` the plan lane REFUSES instead of dispatching

   830:     an operator can see which rung would blow the ceiling.
   831:     """
   832:     node_ceiling = 0.0

   839:             if not isinstance(detail, dict):
   840:                 continue
   841:             route = detail.get("route") if isinstance(detail.get("route"), dict) else {}
   842:             try:
   843:                 node_ceiling += float(route.get("cost_ceiling") or detail.get("cost_ceiling") or 0.0)
   844:             except (TypeError, ValueError):

   847:     decompose_cost = 0.0
   848:     waist_cost = 0.0

   852:             model = decompose_model
   853:             if not model:

   856:                                                  custom_frontier=frontier_model)[0]
   857:                 except Exception:

   860:         if plan_consensus:
   861:             model = decompose_model

   864:                     model = resolve_scout_ladder(use_free=use_free,

   874:             except Exception:

   876:             # Worst-case: every ladder rung could be attempted across the

  1045: def plan_task(
  1046:     goal: str,
  1047:     candidate_files: Optional[Sequence[str]] = None,
  1048:     repo_context: Optional[str] = None,
  1049:     custom_frontier: Optional[str] = None,
  1050:     use_free: bool = True,
  1051:     decomposed_dag: Optional[TaskDAG] = None,
  1052:     root: Optional[str] = None,
  1053:     run_gate: Optional[str] = None,
  1054:     allow_escalation: bool = False,
  1055:     jev_route: Optional[str] = None,
  1056: ) -> Dict[str, Any]:
  1057:     # Formulate a TaskDAG and classify sliding-scale tiers for each node.
  1058:     # decomposed_dag: a pre-built DAG (LLM-authored via decompose_via_llm or
  1059:     # waist-amended) replacing the heuristic decomposition; tier
  1060:     # classification and ceiling math are identical for either origin.
  1061:     # root/run_gate feed the ONE gate rule (repo_scope.gate_for_targets):
  1062:     # every node leaves this function with a verification gate derived from
  1063:     # its OWN declared target, so cli, mcp and the agent lane all dispatch
  1064:     # gated nodes instead of each lane re-deriving the rule (or omitting
  1065:     # it: an ungated write is refused at unknown trust, so the MCP lane's
  1066:     # plan_and_execute could not land anything at all).
  1067:     # jev_route: optional JEV-P3 typed route (vocabulary only) used as a
  1068:     # tier floor when keyed; unkeyed callers leave it None.
  1069:     dag = (decomposed_dag if decomposed_dag is not None
  1070:            else heuristic_decompose_goal(goal, candidate_files,
  1071:                                          root_dir=root))
  1072:     node_details: List[Dict[str, Any]] = []
  1073:     total_ceiling = 0.0
  1074: 
  1075:     classified_nodes: Dict[str, DAGNode] = {}
  1076:     batches = dag.topological_batches()
  1077:     depth_map: Dict[str, int] = {}
  1078:     for depth, batch in enumerate(batches):
  1079:         for n in batch:
  1080:             depth_map[n.node_id] = depth
  1081: 
  1082:     for node_id, node in dag.nodes.items():
  1083:         is_leaf = len(node.dependencies) == 0
  1084:         depth = depth_map.get(node_id, 0)
  1085:         gate = gate_for_targets(node.target_files, root,
  1086:                                declared=node.local_gate, run_gate=run_gate)
  1087:         route = resolve_sliding_scale_route(
  1088:             instruction=node.instruction,
  1089:             target_files=node.target_files,
  1090:             dependency_depth=depth,
  1091:             is_leaf=is_leaf,
  1092:             use_free=use_free,
  1093:             custom_frontier=custom_frontier,
  1094:             allow_escalation=allow_escalation,
  1095:             jev_route=jev_route,
  1096:         )
  1097:         total_ceiling += route.cost_ceiling
  1098:         classified_nodes[node_id] = DAGNode(
  1099:             node_id=node.node_id,
  1100:             instruction=node.instruction,
  1101:             target_files=node.target_files,
  1102:             create_files=node.create_files,

  1144: def node_apply_kwargs(
  1145:     node_detail: Optional[Dict[str, Any]] = None,
  1146:     explicit_model: Optional[str] = None,
  1147:     explicit_task_max_cost: Optional[float] = None,
  1148:     allow_escalation: Optional[bool] = None,
  1149:     attest_model: Optional[str] = None,
  1150: ) -> Dict[str, Any]:
  1151:     """Per-request apply kwargs for executing one planned DAG node.
  1152: 
  1153:     Threads a node's sliding-scale route (from :func:`plan_task`'s
  1154:     ``nodes`` entries) into :meth:`harness.apply.ApplyEngine.apply_edit`:
  1155:     the tier ladder becomes the request's ``apply_pool`` -- the engine
  1156:     orders it at the routing boundary via ``capability.ordered_pool``
  1157:     (interfaces never pre-order a pool) and rotation stays inside the
  1158:     tier-appropriate models. The tier cost ceiling becomes the per-task
  1159:     ceiling only when it is a real bound (paid tiers): a $0 free-tier
  1160:     ceiling is deliberately NOT passed, because a zero task budget would
  1161:     refuse the escalation ladder before it could rescue a hard node
  1162:     (``spent - start < 0.0`` is never true); free-tier cost discipline is
  1163:     the governor's job (every attempted model bills $0.0).
  1164: 
  1165:     An explicit model pin wins outright (manual routing, no pool override);
  1166:     an explicit ``task_max_cost`` pin suppresses only the ceiling.
  1167:     Unknown, missing, or MALFORMED route detail degrades to today's
  1168:     defaults ({}): non-mapping detail/route, a non-sequence ladder (a
  1169:     string would otherwise become per-character "model ids"), and
  1170:     non-finite ceilings are all refused silently -- the governor's
  1171:     key-level ceiling still binds, so degradation never fails open.
  1172: 
  1173:     ``allow_escalation`` and ``attest_model`` are threaded verbatim when the
  1174:     caller supplies them, so a lane whose node can fail a gate carries the
  1175:     escalation arming and the verifier identity explicitly instead of
  1176:     inheriting them silently from the session defaults. ``None`` omits the
  1177:     key, leaving the engine/settings default in charge.
  1178:     """
  1179:     if explicit_model is not None:
  1180:         return {}
  1181:     detail = node_detail if isinstance(node_detail, dict) else {}
  1182:     route = detail.get("route")
  1183:     if not isinstance(route, dict):
  1184:         route = {}
  1185:     kwargs = {}
  1186:     raw_ladder = route.get("ladder")
  1187:     if isinstance(raw_ladder, (list, tuple)):
  1188:         ladder = [str(m).strip() for m in raw_ladder if str(m).strip()]
  1189:         if ladder:
  1190:             kwargs["apply_pool"] = ladder
  1191:     if explicit_task_max_cost is None:
  1192:         try:
  1193:             ceiling = float(route.get("cost_ceiling"))
  1194:         except (TypeError, ValueError):
  1195:             ceiling = 0.0
  1196:         if ceiling > 0.0 and math.isfinite(ceiling):
  1197:             kwargs["task_max_cost"] = ceiling
  1198:     if allow_escalation is not None:
  1199:         kwargs["allow_escalation"] = bool(allow_escalation)
  1200:     if attest_model:
  1201:         kwargs["attest_model"] = str(attest_model)
  1202:     # A chunked large-file node carries the engine's own backend vocabulary;
  1203:     # anything else stays out (the engine validates it again at the boundary).
  1204:     backend = detail.get("backend")
  1205:     if backend in ("harness", "morph", "diff"):
  1206:         kwargs["backend"] = backend
  1207:     return kwargs

  1212: def build_waist_prompt(
  1213:     plan_result: Dict[str, Any],
  1214:     brief_context: str = "",
  1215:     window_context: str = "",
  1216:     consensus: Optional[Dict[str, Any]] = None,
  1217: ) -> str:
  1218:     """Build the frontier waist-confirmation prompt (M2).
  1219: 
  1220:     The brief (file signatures + bounded windows + failure evidence) is the
  1221:     confirming model's ONLY repo access: bounded file-window round-trips
  1222:     replace open-ended reading. The verdict contract is strict JSON, one of::
  1223: 
  1224:         {"verdict": "approve"}
  1225:         {"verdict": "amend", "nodes": [ ...same node schema as decomposition... ]}
  1226:         {"verdict": "refuse", "reason": "...", "evidence": "cited brief section"}

  1231:     amend), and the replacement re-validates through the same schema as the
  1232:     original plan. A refusal must cite the brief section that fails --
  1233:     honest evidence, not vibes.
  1234:     """
  1235:     nodes = [
  1236:         {k: n[k] for k in ("node_id", "instruction", "target_files",
  1237:                            "create_files",
  1238:                            "dependencies", "local_gate", "complexity_tier")
  1239:          if k in n}
  1240:         for n in plan_result.get("nodes", [])
  1241:     ]
  1242:     lines = [
  1243:         "You are the plan-confirmation gate for an autonomous coding harness.",
  1244:         "A cheaper model decomposed the goal below into an executable DAG.",
  1245:         "Confirm the plan BEFORE execution spend: check decomposition quality,",
  1246:         "tier assignments (0=scout/simple, 1=standard, 2=deep/frontier),",
  1247:         "dependency ordering, and target-file scoping.",
  1248:         "",
  1249:         "GOAL:",
  1250:         plan_result.get("goal", "").strip(),
  1251:         "",
  1252:         "PLANNED DAG (JSON):",
  1253:         json.dumps({"nodes": nodes}, indent=2),
  1254:         "",
  1255:     ]
  1256:     if brief_context:
  1257:         lines.extend(["REPOSITORY BRIEF (signatures; your only repo access):",
  1258:                       brief_context.strip(), ""])
  1259:     if window_context:
  1260:         lines.extend(["REQUESTED FILE WINDOWS (attached this round):",
  1261:                       window_context.strip(), ""])
  1262:     if consensus is not None and not consensus.get("sound", True):
  1263:         lines.extend([
  1264:             "PLAN CONSENSUS (cheap pre-check marked this plan UNSOUND):",
  1265:             json.dumps({"sound": False,
  1266:                         "reasons": list(consensus.get("reasons") or [])}, indent=2),
  1267:             "You MUST return 'amend' with a corrected replacement DAG. Do not",
  1268:             "approve an unsound plan.",
  1269:             "",
  1270:         ])
  1271:     lines.extend([
  1272:         "VERDICT CONTRACT -- respond with ONLY one JSON object:",
  1273:         '  {"verdict": "approve"}                                   plan is sound as routed',
  1274:         '  {"verdict": "amend", "nodes": [...]}                     full replacement node set,',
  1275:         '                                                           same schema, re-validated;',
  1276:         '                                                           subdividing a node is an amend',
  1277:         '  {"verdict": "refuse", "reason": "...", "evidence": "..."}  cite the brief section',
  1278:         '                                                           that fails; evidence required',
  1279:         '  {"verdict": "request_windows", "file_window_requests": [...]}  need bounded source',
  1280:         '                                                           windows before deciding',
  1281:         '                                                            ({"path", "start_line",',
  1282:         '                                                              "end_line"}; 1-based,',
  1283:         '                                                              end inclusive)',
  1284:         "Rules: the DAG must stay acyclic; node instructions must be precise and",
  1285:         "scoped (they are executed verbatim by cheaper models); do not request",
  1286:         "more windows than you need -- round-trips are budgeted.",
  1287:         "AMENDMENT vs REFUSAL POLICY:",
  1288:         "- DO NOT REFUSE simply because the planned DAG is incomplete, lacks iteration loops,",
  1289:         "  has missing steps, or needs different granularity/ordering. If the planned DAG is",
  1290:         "  suboptimal, REPAIR IT: return 'amend' with the complete, corrected replacement DAG nodes.",
  1291:         "- 'refuse' is STRICTLY reserved for requests that are genuinely impossible, out of scope,",
  1292:         "  destructive, or malicious. Refusing an executable coding task is a failure.",
  1293:     ])
  1294:     return "\n".join(lines)

  1298: def _waist_window_request(raw) -> Dict[str, Any]:
  1299:     if not isinstance(raw, dict):
  1300:         raise HarnessError("waist file_window_requests entries must be objects")
  1301:     path = str(raw.get("path") or "").strip()
  1302:     if not path:
  1303:         raise HarnessError("waist file_window_request missing 'path'")
  1304:     request: Dict[str, Any] = {"path": path}
  1305:     for key in ("start_line", "end_line"):
  1306:         value = raw.get(key)
  1307:         if value is None:
  1308:             continue
  1309:         try:
  1310:             line = int(value)
  1311:         except (TypeError, ValueError):
  1312:             raise HarnessError(
  1313:                 f"waist file_window_request {key} must be an integer") from None
  1314:         if line < 1:
  1315:             raise HarnessError(f"waist file_window_request {key} must be >= 1")
  1316:         request[key] = line
  1317:     if ("start_line" in request or "end_line" in request) and \
  1318:             request.get("start_line", 1) > request.get("end_line", 1 << 30):
  1319:         raise HarnessError("waist file_window_request start_line exceeds end_line")
  1320:     return request

  1341: def _canonical_verdict(data: Dict[str, Any]) -> str:
  1342:     """The canonical verdict kind for a parsed verdict body.
  1343: 
  1344:     Tolerates the spelling and envelope differences models produce --
  1345:     ``APPROVED``, ``Approved.``, ``{"result": {"verdict": ...}}`` -- while
  1346:     keeping the decision set closed. An unrecognised or absent verdict
  1347:     returns ``""`` and the caller fails closed.
  1348:     """
  1349:     body = data
  1350:     if "verdict" not in body:
  1351:         # One level of envelope: {"result": {...}} / {"plan": {...}} / ...
  1352:         for key in ("result", "plan", "response", "output", "decision"):
  1353:             nested = body.get(key)
  1354:             if isinstance(nested, dict) and "verdict" in nested:
  1355:                 body = nested
  1356:                 break
  1357:     raw = body.get("verdict")
  1358:     if isinstance(raw, dict):
  1359:         raw = raw.get("verdict") or raw.get("kind") or raw.get("decision")
  1360:     # Models append their own sentence punctuation: "Approved.", "Approve!".
  1361:     key = str(raw or "").strip().strip(".!?").strip().lower().replace(" ", "_")
  1362:     return _VERDICT_ALIASES.get(key, "")

  1365: def parse_waist_verdict(response_text: str) -> Dict[str, Any]:
  1366:     """Parse and validate a waist verdict (strict; fail-closed).
  1367: 
  1368:     Returns one of ``{"verdict": "approve"}``, ``{"verdict": "amend",
  1369:     "dag": TaskDAG}`` (the replacement node set, re-validated for unique
  1370:     ids, unknown dependencies, cycles, and apply-time instruction
  1371:     length), ``{"verdict": "split", "dag": TaskDAG}`` (same shape and
  1372:     re-validation as amend; recorded distinctly so a planner that
  1373:     subdivides a node gets its own ledgered kind -- MR-4's four-way
  1374:     enum), ``{"verdict": "refuse", "reason", "evidence"}``, or
  1375:     ``{"verdict": "request_windows", "file_window_requests": [...]}``.
  1376:     """
  1377:     data = _parse_json_object(response_text, "waist plan verdict")
  1378:     verdict = _canonical_verdict(data)
  1379:     if verdict == "approve":
  1380:         return {"verdict": "approve"}
  1381:     if verdict in ("amend", "split"):
  1382:         nodes = data.get("nodes")
  1383:         if not isinstance(nodes, list) or not nodes:
  1384:             raise HarnessError(
  1385:                 f"waist {verdict} verdict requires a non-empty 'nodes' list")
  1386:         amended = TaskDAG.from_dict({"nodes": nodes})
  1387:         for node in amended.nodes.values():
  1388:             if len(node.instruction) > MAX_INSTRUCTION_CHARS:
  1389:                 raise HarnessError(
  1390:                     f"amended node {node.node_id!r} instruction exceeds "
  1391:                     f"{MAX_INSTRUCTION_CHARS} chars (would fail apply validation)")
  1392:         return {"verdict": verdict, "dag": amended}
  1393:     if verdict == "refuse":
  1394:         reason = str(data.get("reason") or "").strip()
  1395:         evidence = str(data.get("evidence") or "").strip()
  1396:         if not reason or not evidence:
  1397:             raise HarnessError(
  1398:                 "waist refuse verdict requires non-empty 'reason' and "
  1399:                 "'evidence' (cite the brief section that fails)")
  1400:         return {"verdict": "refuse", "reason": reason, "evidence": evidence}
  1401:     if verdict == "request_windows":
  1402:         requests = data.get("file_window_requests")
  1403:         if not isinstance(requests, list) or not requests:
  1404:             raise HarnessError(
  1405:                 "waist request_windows verdict requires a non-empty "
  1406:                 "'file_window_requests' list")
  1407:         return {"verdict": "request_windows",
  1408:                 "file_window_requests": [_waist_window_request(r) for r in requests]}
  1409:     raw = str((data.get("verdict") if not isinstance(data.get("verdict"), dict)
  1410:                else data["verdict"].get("verdict")) or "").strip()
  1411:     raise HarnessError(f"unknown waist verdict {raw or '<missing>'!r}")

  1422: def read_window(path: str, start_line=None, end_line=None) -> str:
  1423:     """Bounded source-window read for the brief (cwd-contained).
  1424: 
  1425:     Model-chosen paths are contained to the working tree (absolute paths
  1426:     and traversal outside the cwd are refused in-band, as an honest
  1427:     "refused" note the model can react to); missing files answer in-band
  1428:     too -- the window round-trip stays cheap and non-fatal.
  1429:     """
  1430:     raw = str(path)
  1431:     refused = "WINDOW REFUSED: only paths inside the working tree are readable"
  1432:     if os.path.isabs(raw) or ".." in raw.split(os.sep) or ".." in raw.split("/"):
  1433:         return refused
  1434:     root = os.path.realpath(os.getcwd())
  1435:     full = os.path.realpath(os.path.join(root, raw))
  1436:     if full != root and not full.startswith(root + os.sep):
  1437:         return refused
  1438:     try:
  1439:         with open(full, encoding="utf-8", errors="replace") as f:
  1440:             lines = f.readlines()
  1441:     except OSError:
  1442:         return f"FILE NOT FOUND: {raw}"
  1443:     total = len(lines)
  1444:     start = max(int(start_line or 1), 1)
  1445:     if start > total:
  1446:         return f"WINDOW EMPTY: {raw} has {total} lines (requested start {start})"
  1447:     end = min(int(end_line or total), total, start + MAX_WINDOW_LINES - 1)
  1448:     body = "".join(lines[start - 1:end])
  1449:     return f"--- WINDOW: {raw} lines {start}-{end} of {total} ---\n{body}"

  1477: def confirm_plan(*, transport, api_key, governor, ledger, plan_result,
  1478:                  model, use_free=True, custom_frontier=None,
  1479:                  chat_fn=None, reader=read_window,
  1480:                  root=None, run_gate=None,
  1481:                  consensus=None,
  1482:                  max_rounds=MAX_WAIST_ROUNDS) -> Dict[str, Any]:
  1483:     """Run the waist: frontier confirms/repairs the plan; return the plan.
  1484: 
  1485:     Returns the (possibly amended) plan_result with a ``confirmation``
  1486:     field, or a ``status: "refused"`` envelope when the model refuses
  1487:     (callers must not execute it -- ``terminal_exit_code`` maps the
  1488:     unknown status to 2 fail-closed). Raises HarnessError when the model
  1489:     keeps requesting windows past ``max_rounds`` (the plan stays
  1490:     unconfirmed and unexecuted), on any transport/parse failure, or when
  1491:     ``model`` is unset (confirmation cannot silently skip itself).
  1492: 
  1493:     ``chat_fn(prompt) -> (text, cost)`` and ``reader(path, start, end) ->
  1494:     str`` are injectable seams for hermetic tests.
  1495:     """
  1496:     if not model:
  1497:         raise HarnessError(
  1498:             "waist confirmation requires a frontier model "
  1499:             "(--frontier-model or HARNESS_FRONTIER_MODEL)")
  1500:     if chat_fn is None:
  1501:         def chat_fn(prompt):
  1502:             return governed_text(transport, api_key, governor, model, prompt,
  1503:                                  WAIST_MAX_TOKENS, label="waist")
  1504:     brief = _brief_for(plan_result)
  1505:     task_id = plan_task_id(plan_result)
  1506:     if ledger is not None:
  1507:         ledger.ensure_offered_task_id(
  1508:             task_id, model=model, required=False, source="plan_confirmation")
  1509:     window_context = ""
  1510:     spent_before = governor.spent if governor is not None else 0.0
  1511: 
  1512:     for round_no in range(1, max_rounds + 1):
  1513:         prompt = build_waist_prompt(plan_result, brief, window_context,
  1514:                                     consensus=consensus)
  1515:         text, _cost = chat_fn(prompt)
  1516:         try:
  1517:             verdict = parse_waist_verdict(text)
  1518:         except HarnessError:
  1519:             # Fail closed on a malformed verdict: the spend already happened,
  1520:             # the plan stays unconfirmed, and the raw body is the evidence.
  1521:             if ledger is not None:
  1522:                 ledger.append("plan_verdict", task_id=task_id, verdict="unparseable",
  1523:                               model=model, round=round_no)
  1524:             raise
  1525: 
  1526:         kind = verdict["verdict"]
  1527:         cost = _verdict_cost(governor, spent_before)
  1528:         if kind == "approve":
  1529:             plan_result["confirmation"] = {
  1530:                 "verdict": "approved", "model": model, "rounds": round_no,
  1531:                 "cost": cost}
  1532:             if ledger is not None:
  1533:                 ledger.append("plan_verdict", task_id=task_id, verdict="approved",
  1534:                               model=model, rounds=round_no, cost=cost)
  1535:             return plan_result
  1536:         if kind == "refuse":
  1537:             refused = dict(plan_result)
  1538:             refused["status"] = "refused"
  1539:             refused["confirmation"] = {
  1540:                 "verdict": "refused", "model": model, "rounds": round_no,
  1541:                 "reason": verdict["reason"], "evidence": verdict["evidence"],
  1542:                 "cost": cost}
  1543:             if ledger is not None:
  1544:                 ledger.append("plan_verdict", task_id=task_id, verdict="refused",
  1545:                               model=model, rounds=round_no, cost=cost,
  1546:                               reason=verdict["reason"])
  1547:             return refused
  1548:         if kind in ("amend", "split"):
  1549:             amended = plan_task(
  1550:                 plan_result.get("goal", ""),
  1551:                 custom_frontier=custom_frontier, use_free=use_free,
  1552:                 decomposed_dag=verdict["dag"], root=root, run_gate=run_gate)
  1553:             confirmed_kind = "amended" if kind == "amend" else "split"
